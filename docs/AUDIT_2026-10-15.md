### Step 1: Supabase, read-only queries (founder runs them; paste the result rows)

**Block A: 029 check**
```sql
select
  (select count(*) from information_schema.columns where table_schema='public'
     and table_name='discovered_companies' and column_name in
     ('watching','approved_at','rejected_at','alive','last_alive_at','quiet_since')) as company_columns,
  (select count(*) from information_schema.columns where table_schema='public'
     and table_name='registry_seeds' and column_name in ('hidden_at','hidden_reason')) as seed_columns,
  (select count(*) from public.discovered_companies where watching) as watching,
  (select count(*) from public.registry_seeds where hidden_at is not null) as hidden;
```

**Block B: Cori counts**
```sql
select
  (select count(*) from public.discovery_listings) as listings,
  (select count(*) from public.discovered_companies) as companies,
  (select count(*) from public.discovered_companies where site_ok) as sites_ok,
  (select count(*) from public.discovered_companies where watching) as watching,
  (select count(*) from public.discovered_companies where watching and alive) as watching_alive,
  (select count(*) from public.discovered_companies where quiet_since is not null) as went_quiet,
  (select count(*) from public.discovered_companies where approved_at is not null) as approved,
  (select count(*) from public.discovered_companies where rejected_at is not null) as rejected;
```

**Block C: Cori runs in the last 7 days**
```sql
select kind, count(*) as runs, count(*) filter (where ok) as ok_runs,
       count(*) filter (where ok is false) as failed, max(started_at) as last_run
from public.cori_runs where started_at > now() - interval '7 days'
group by kind order by kind;
```

**Block D: Monitoring checks in the last 7 days**
```sql
select check_type, status, count(*) as checks
from public.checks where started_at > now() - interval '7 days'
group by check_type, status order by check_type, status;
```

**Block E: Incidents in the last 30 days**
```sql
select status, count(*) as incidents, max(opened_at) as latest
from public.incidents where opened_at > now() - interval '30 days'
group by status;
```

**Block F: Spending (passed paid checks)**
```sql
select
  coalesce(sum(observed_price) filter (where started_at >= date_trunc('day', now())), 0)   as spent_today,
  coalesce(sum(observed_price) filter (where started_at >= date_trunc('month', now())), 0) as spent_month,
  coalesce(sum(observed_price) filter (where started_at > now() - interval '7 days'), 0)   as spent_7d
from public.checks where status = 'passed' and observed_price is not null;
```

**Block G: Services and owners, plus database size**
```sql
select
  (select count(*) from public.services where deleted_at is null) as active_services,
  (select count(distinct user_id) from public.services where deleted_at is null) as owners,
  (select count(*) from public.profiles) as accounts,
  pg_size_pretty(pg_database_size(current_database())) as db_size;
```
